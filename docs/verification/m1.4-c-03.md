```powershell
Set-Location 'E:\Project\CampusHub\miniprogram\dist'
$compilerRoot = 'E:\CampusHubRuntime\wechat-devtools\resources\app.asar.unpacked\node_modules\wcc-exec'
$wxml = @(Get-ChildItem -Recurse -File -Filter '*.wxml' | ForEach-Object { [System.IO.Path]::GetRelativePath((Get-Location).Path, $_.FullName).Replace('\', '/') })
$wxss = @(Get-ChildItem -Recurse -File -Filter '*.wxss' | ForEach-Object { [System.IO.Path]::GetRelativePath((Get-Location).Path, $_.FullName).Replace('\', '/') })
& (Join-Path $compilerRoot 'wcc.exe') -o (Join-Path $env:TEMP 'campushub-c03-wxml.js') @wxml
if ($LASTEXITCODE -ne 0) { throw 'WXML compile failed' }
& (Join-Path $compilerRoot 'wcsc.exe') -lc -o (Join-Path $env:TEMP 'campushub-c03-wxss.js') @wxss
if ($LASTEXITCODE -ne 0) { throw 'WXSS compile failed' }
```
